## Array

### 1. Two Sum
Find indices of two numbers that add to `target`.
```python
# Brute: check all pairs
def two_sum_brute(nums, target):
    for i in range(len(nums)):
        for j in range(i+1, len(nums)):
            if nums[i] + nums[j] == target:
                return [i, j]
```
**TC:** O(n²) **SC:** O(1)
```python
# Optimal: hashmap value -> index
def two_sum(nums, target):
    seen = {}
    for i, x in enumerate(nums):
        if target - x in seen:
            return [seen[target - x], i]
        seen[x] = i
```
**TC:** O(n) **SC:** O(n)

### 2. Best Time to Buy and Sell Stock
Max profit from one buy + one sell.
```python
# Brute: all pairs
def max_profit_brute(prices):
    best = 0
    for i in range(len(prices)):
        for j in range(i+1, len(prices)):
            best = max(best, prices[j] - prices[i])
    return best
```
**TC:** O(n²) **SC:** O(1)
```python
# Optimal: track min price so far
def max_profit(prices):
    min_p, best = float("inf"), 0
    for p in prices:
        min_p = min(min_p, p)
        best = max(best, p - min_p)
    return best
```
**TC:** O(n) **SC:** O(1)

### 3. Contains Duplicate

```python
# Brute: nested comparison
def contains_dup_brute(nums):
    for i in range(len(nums)):
        for j in range(i+1, len(nums)):
            if nums[i] == nums[j]:
                return True
    return False
```
**TC:** O(n²) **SC:** O(1)
```python
# Optimal: set membership
def contains_dup(nums):

    for x in nums:

```

### 4. Product of Array Except Self
Output array where `out[i]` = product of all elements except `nums[i]`, no division.
```python
# Brute: recompute product each time (division-free but slow)
def product_except_self_brute(nums):
    res = []
    for i in range(len(nums)):
        p = 1
        for j in range(len(nums)):
            if i != j:
                p *= nums[j]
        res.append(p)
    return res
```
**TC:** O(n²) **SC:** O(1) extra (excluding output)
```python
# Optimal: prefix * suffix products
def product_except_self(nums):
    n = len(nums)
    res = [1] * n
    prefix = 1
    for i in range(n):
        res[i] = prefix
        prefix *= nums[i]
    suffix = 1
    for i in range(n-1, -1, -1):
        res[i] *= suffix
        suffix *= nums[i]
    return res
```
**TC:** O(n) **SC:** O(1) extra (excluding output)

### 5. Maximum Subarray
```python
# Brute: all subarrays
def max_subarray_brute(nums):
    best = float("-inf")
    for i in range(len(nums)):
        cur = 0
        for j in range(i, len(nums)):
            cur += nums[j]
            best = max(best, cur)
    return best
```
**TC:** O(n²) **SC:** O(1)
```python
# Optimal: Kadane's
def max_subarray(nums):
    best = cur = nums[0]
    for x in nums[1:]:
        cur = max(x, cur + x)
        best = max(best, cur)
    return best
```
**TC:** O(n) **SC:** O(1)

### 6. Maximum Product Subarray
```python
# Brute: all subarrays
def max_product_brute(nums):
    best = nums[0]
    for i in range(len(nums)):
        p = 1
        for j in range(i, len(nums)):
            p *= nums[j]
            best = max(best, p)
    return best
```
**TC:** O(n²) **SC:** O(1)
```python
# Optimal: track running max & min (negatives flip them)
def max_product(nums):
    best = cur_max = cur_min = nums[0]
    for x in nums[1:]:
        candidates = (x, cur_max * x, cur_min * x)
        cur_max, cur_min = max(candidates), min(candidates)
        best = max(best, cur_max)
    return best
```
**TC:** O(n) **SC:** O(1)

### 7. Find Minimum in Rotated Sorted Array
```python
# Brute: linear scan
def find_min_brute(nums):
    return min(nums)
```
**TC:** O(n) **SC:** O(1)
```python
# Optimal: binary search on rotation point
def find_min(nums):
    lo, hi = 0, len(nums) - 1
    while lo < hi:
        mid = (lo + hi) // 2
        if nums[mid] > nums[hi]:
            lo = mid + 1
        else:
            hi = mid
    return nums[lo]
```
**TC:** O(log n) **SC:** O(1)

### 8. Search in Rotated Sorted Array
```python
# Brute: linear scan
def search_brute(nums, target):
    for i, x in enumerate(nums):
        if x == target:
            return i
    return -1
```
**TC:** O(n) **SC:** O(1)
```python
# Optimal: modified binary search
def search(nums, target):
    lo, hi = 0, len(nums) - 1
    while lo <= hi:
        mid = (lo + hi) // 2
        if nums[mid] == target:
            return mid
        if nums[lo] <= nums[mid]:
            if nums[lo] <= target < nums[mid]:
                hi = mid - 1
            else:
                lo = mid + 1
        else:
            if nums[mid] < target <= nums[hi]:
                lo = mid + 1
            else:
                hi = mid - 1
    return -1
```
**TC:** O(log n) **SC:** O(1)

### 9. 3Sum
All unique triplets summing to 0.
```python
# Brute: three nested loops + set for dedup
def three_sum_brute(nums):
    n, res = len(nums), set()
    for i in range(n):
        for j in range(i+1, n):
            for k in range(j+1, n):
                if nums[i]+nums[j]+nums[k] == 0:
                    res.add(tuple(sorted((nums[i], nums[j], nums[k]))))
    return [list(t) for t in res]
```
**TC:** O(n³) **SC:** O(n) for result set
```python
# Optimal: sort + two pointers per fixed first element
def three_sum(nums):
    nums.sort()
    res = []
    for i in range(len(nums) - 2):
        if i > 0 and nums[i] == nums[i-1]:
            continue
        lo, hi = i+1, len(nums)-1
        while lo < hi:
            s = nums[i] + nums[lo] + nums[hi]
            if s < 0:
                lo += 1
            elif s > 0:
                hi -= 1
            else:
                res.append([nums[i], nums[lo], nums[hi]])
                lo += 1
                while lo < hi and nums[lo] == nums[lo-1]:
                    lo += 1
    return res
```
**TC:** O(n²) **SC:** O(1) extra (excluding sort/output)

### 10. Container With Most Water
```python
# Brute: all pairs of lines
def max_area_brute(height):
    best = 0
    for i in range(len(height)):
        for j in range(i+1, len(height)):
            best = max(best, min(height[i], height[j]) * (j - i))
    return best
```
**TC:** O(n²) **SC:** O(1)
```python
# Optimal: two pointers, move the shorter side inward
def max_area(height):
    lo, hi, best = 0, len(height) - 1, 0
    while lo < hi:
        best = max(best, min(height[lo], height[hi]) * (hi - lo))
        if height[lo] < height[hi]:
            lo += 1
        else:
            hi -= 1
    return best
```
**TC:** O(n) **SC:** O(1)

## Binary

### 11. Sum of Two Integers
Add two integers without `+`/`-`.
```python
# Brute (not truly bitwise, but a "naive" fallback using Python big ints)
def get_sum_brute(a, b):
    while b != 0:            # still bit logic — Python has no fixed-width overflow
        carry = a & b
        a = a ^ b
        b = carry << 1
    return a
```
**TC:** O(1) (bounded by 32 bits) **SC:** O(1)
```python
# Optimal: same bitwise approach, masked to simulate 32-bit overflow correctly
def get_sum(a, b):
    mask = 0xFFFFFFFF
    while b & mask:
        a, b = (a ^ b) & mask, ((a & b) << 1) & mask
    return a if a <= 0x7FFFFFFF else ~(a ^ mask)
```
**TC:** O(1) **SC:** O(1)

### 12. Number of 1 Bits
```python
# Brute: string conversion
def hamming_weight_brute(n):
    return bin(n).count("1")
```
**TC:** O(32) = O(1) **SC:** O(1)
```python
# Optimal: Brian Kernighan's algorithm — clears lowest set bit each time
def hamming_weight(n):
    count = 0
    while n:
        n &= (n - 1)
        count += 1
    return count
```
**TC:** O(number of set bits) ≤ O(32) **SC:** O(1)

### 13. Counting Bits
For `0..n`, count set bits of each number.
```python
# Brute: count bits independently for each number
def count_bits_brute(n):
    return [bin(i).count("1") for i in range(n + 1)]
```
**TC:** O(n log n) **SC:** O(n) for output
```python
# Optimal: DP — i's bit count = (i >> 1)'s count + (i & 1)
def count_bits(n):
    dp = [0] * (n + 1)
    for i in range(1, n + 1):
        dp[i] = dp[i >> 1] + (i & 1)
    return dp
```
**TC:** O(n) **SC:** O(n) for output

### 14. Missing Number
Array has n distinct numbers from `0..n`, one missing.
```python
# Brute: check membership of every expected value
def missing_number_brute(nums):
    for i in range(len(nums) + 1):
        if i not in nums:
            return i
```
**TC:** O(n²) **SC:** O(1)
```python
# Optimal: Gauss sum formula difference (or XOR)
def missing_number(nums):
    n = len(nums)
    return n * (n + 1) // 2 - sum(nums)
```
**TC:** O(n) **SC:** O(1)

### 15. Reverse Bits
Reverse bits of a 32-bit unsigned integer.
```python
# Brute: string reverse
def reverse_bits_brute(n):
    return int(bin(n)[2:].zfill(32)[::-1], 2)
```
**TC:** O(32) = O(1) **SC:** O(32) = O(1)
```python
# Optimal: bit-shifting accumulator, no string conversion
def reverse_bits(n):
    result = 0
    for _ in range(32):
        result = (result << 1) | (n & 1)
        n >>= 1
    return result
```
**TC:** O(32) = O(1) **SC:** O(1)

## Dynamic Programming

### 16. Climbing Stairs
```python
# Brute: recursion, recomputes overlapping subproblems
def climb_stairs_brute(n):
    if n <= 2:
        return n
    return climb_stairs_brute(n-1) + climb_stairs_brute(n-2)
```
**TC:** O(2ⁿ) **SC:** O(n) recursion stack
```python
# Optimal: bottom-up, two running variables (Fibonacci pattern)
def climb_stairs(n):
    if n <= 2:
        return n
    a, b = 1, 2
    for _ in range(3, n + 1):
        a, b = b, a + b
    return b
```
**TC:** O(n) **SC:** O(1)

### 17. Coin Change
Fewest coins to make `amount`.
```python
# Brute: try every coin recursively
def coin_change_brute(coins, amount):
    if amount == 0:
        return 0
    if amount < 0:
        return float("inf")
    best = float("inf")
    for c in coins:
        best = min(best, coin_change_brute(coins, amount - c) + 1)
    return best
```
**TC:** O(coins^amount) **SC:** O(amount) recursion stack
```python
# Optimal: bottom-up DP
def coin_change(coins, amount):
    dp = [0] + [float("inf")] * amount
    for a in range(1, amount + 1):
        for c in coins:
            if c <= a:
                dp[a] = min(dp[a], dp[a - c] + 1)
    return dp[amount] if dp[amount] != float("inf") else -1
```
**TC:** O(amount · len(coins)) **SC:** O(amount)

### 18. Longest Increasing Subsequence
```python
# Brute: DP, compare against every earlier element
def length_of_lis_brute(nums):
    n = len(nums)
    dp = [1] * n
    for i in range(n):
        for j in range(i):
            if nums[j] < nums[i]:
                dp[i] = max(dp[i], dp[j] + 1)
    return max(dp) if dp else 0
```
**TC:** O(n²) **SC:** O(n)
```python
# Optimal: binary search on a "tails" array
import bisect
def length_of_lis(nums):
    tails = []
    for x in nums:
        i = bisect.bisect_left(tails, x)
        if i == len(tails):
            tails.append(x)
        else:
            tails[i] = x
    return len(tails)
```
**TC:** O(n log n) **SC:** O(n)

### 19. Longest Common Subsequence
```python
# Brute: recursive without memoization
def lcs_brute(a, b, i=0, j=0):
    if i == len(a) or j == len(b):
        return 0
    if a[i] == b[j]:
        return 1 + lcs_brute(a, b, i+1, j+1)
    return max(lcs_brute(a, b, i+1, j), lcs_brute(a, b, i, j+1))
```
**TC:** O(2^(n+m)) **SC:** O(n+m) recursion stack
```python
# Optimal: bottom-up DP table
def lcs(a, b):
    n, m = len(a), len(b)
    dp = [[0] * (m + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        for j in range(1, m + 1):
            if a[i-1] == b[j-1]:
                dp[i][j] = dp[i-1][j-1] + 1
            else:
                dp[i][j] = max(dp[i-1][j], dp[i][j-1])
    return dp[n][m]
```
**TC:** O(n·m) **SC:** O(n·m)

### 20. Word Break
Can `s` be segmented into words from `wordDict`?
```python
# Brute: try every split point recursively
def word_break_brute(s, wordDict, start=0):
    if start == len(s):
        return True
    for end in range(start + 1, len(s) + 1):
        if s[start:end] in wordDict and word_break_brute(s, wordDict, end):
            return True
    return False
```
**TC:** O(2ⁿ) **SC:** O(n) recursion stack
```python
# Optimal: bottom-up DP
def word_break(s, wordDict):
    words = set(wordDict)
    n = len(s)
    dp = [False] * (n + 1)
    dp[0] = True
    for i in range(1, n + 1):
        for j in range(i):
            if dp[j] and s[j:i] in words:
                dp[i] = True
                break
    return dp[n]
```
**TC:** O(n²) **SC:** O(n)

### 21. Combination Sum IV
Count combinations (order matters) that sum to `target`.
```python
# Brute: recursion without memo
def combination_sum4_brute(nums, target):
    if target == 0:
        return 1
    if target < 0:
        return 0
    return sum(combination_sum4_brute(nums, target - n) for n in nums)
```
**TC:** exponential **SC:** O(target) recursion stack
```python
# Optimal: bottom-up DP (like coin change but counting permutations)
def combination_sum4(nums, target):
    dp = [1] + [0] * target
    for t in range(1, target + 1):
        for n in nums:
            if n <= t:
                dp[t] += dp[t - n]
    return dp[target]
```
**TC:** O(target · len(nums)) **SC:** O(target)

### 22. House Robber
Max sum with no two adjacent elements chosen.
```python
# Brute: recursion without memo
def rob_brute(nums, i=0):
    if i >= len(nums):
        return 0
    return max(rob_brute(nums, i+1), nums[i] + rob_brute(nums, i+2))
```
**TC:** O(2ⁿ) **SC:** O(n) recursion stack
```python
# Optimal: two running variables
def rob(nums):
    prev, cur = 0, 0
    for x in nums:
        prev, cur = cur, max(cur, prev + x)
    return cur
```
**TC:** O(n) **SC:** O(1)

### 23. House Robber II
Same, but houses are arranged in a circle.
```python
# Brute: try all subsets, skip invalid (adjacent / wraparound) ones
from itertools import combinations
def rob2_brute(nums):
    n = len(nums)
    best = 0
    for r in range(n + 1):
        for combo in combinations(range(n), r):
            valid = all((combo[i+1] - combo[i]) > 1 for i in range(len(combo)-1))
            if valid and not (0 in combo and n-1 in combo):
                best = max(best, sum(nums[i] for i in combo))
    return best
```
**TC:** O(2ⁿ) **SC:** O(n)
```python
# Optimal: run linear House Robber twice (exclude first, exclude last)
def rob2(nums):
    if len(nums) == 1:
        return nums[0]
    def rob_linear(houses):
        prev, cur = 0, 0
        for x in houses:
            prev, cur = cur, max(cur, prev + x)
        return cur
    return max(rob_linear(nums[1:]), rob_linear(nums[:-1]))
```
**TC:** O(n) **SC:** O(1)

### 24. Decode Ways
Count ways to decode a digit string (A=1..Z=26).
```python
# Brute: recursion without memo
def num_decodings_brute(s, i=0):
    if i == len(s):
        return 1
    if s[i] == '0':
        return 0
    count = num_decodings_brute(s, i+1)
    if i+1 < len(s) and int(s[i:i+2]) <= 26:
        count += num_decodings_brute(s, i+2)
    return count
```
**TC:** O(2ⁿ) **SC:** O(n) recursion stack
```python
# Optimal: bottom-up DP
def num_decodings(s):
    n = len(s)
    dp = [0] * (n + 1)
    dp[0] = 1
    dp[1] = 1 if s[0] != '0' else 0
    for i in range(2, n + 1):
        if s[i-1] != '0':
            dp[i] += dp[i-1]
        if 10 <= int(s[i-2:i]) <= 26:
            dp[i] += dp[i-2]
    return dp[n]
```
**TC:** O(n) **SC:** O(n) (reducible to O(1) with two variables)

### 25. Unique Paths
Robot on an m×n grid, count paths from top-left to bottom-right.

```python
# Optimal: Union-Find — a valid tree has exactly n-1 edges and no cycle
def valid_tree(n, edges):
    if len(edges) != n - 1:
        return False
    parent = list(range(n))
    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x
    for a, b in edges:
        ra, rb = find(a), find(b)
```

```python
    stack = []
    node = root
    while stack or node:
        while node:
            stack.append(node)
            node = node.left
        node = stack.pop()
        k -= 1
        if k == 0:
            return node.val
        node = node.right
```
**TC:** O(h + k) — better than full O(n) traversal when k is small **SC:** O(h)

### 70. Lowest Common Ancestor of BST
```python
# Brute: find full root-to-node paths for both, compare
def lca_brute(root, p, q):
    def find_path(node, target, path):
        if not node:
            return False
        path.append(node)
        if node == target:
            return True
        if find_path(node.left, target, path) or find_path(node.right, target, path):
            return True
        path.pop()
        return False
    path_p, path_q = [], []
    find_path(root, p, path_p)
    find_path(root, q, path_q)
    lca = None
    for a, b in zip(path_p, path_q):
        if a == b:
            lca = a
        else:
            break
    return lca
```
**TC:** O(n) **SC:** O(n) — two full paths stored
```python
# Optimal: use BST ordering property, single traversal
def lca(root, p, q):
    node = root
    while node:
        if p.val < node.val and q.val < node.val:
            node = node.left
        elif p.val > node.val and q.val > node.val:
            node = node.right
        else:
            return node
```

```python
# Optimal: tree of character nodes, O(k) per operation (k = word/prefix length)
class TrieNode:
    def __init__(self):
        self.children = {}
        self.is_end = False

class Trie:
    def __init__(self):
        self.root = TrieNode()
    def insert(self, word):
        node = self.root
        for ch in word:
            node = node.children.setdefault(ch, TrieNode())
        node.is_end = True
    def search(self, word):
        node = self._find(word)
        return node is not None and node.is_end
    def starts_with(self, prefix):
        return self._find(prefix) is not None
    def _find(self, s):
        node = self.root
        for ch in s:
            if ch not in node.children:
                return None
            node = node.children[ch]
        return node
```
**TC:** O(k) per operation **SC:** O(total chars across all words)

### 72. Add and Search Word
Supports `.` wildcard matching any character.
```python
# Brute: store words in a list, brute-match wildcard char by char
class WordDictionaryBrute:
    def __init__(self):
        self.words = []
    def add_word(self, word):
        self.words.append(word)
    def search(self, word):
        for w in self.words:

                continue
            if all(a == b or b == '.' for a, b in zip(w, word)):

        return False
```
**TC:** add O(1), search O(n·k) **SC:** O(total chars)
```python
# Optimal: Trie + DFS for wildcard branching
class TrieNode:
    def __init__(self):
        self.children = {}
        self.is_end = False

class WordDictionary:
    def __init__(self):
        self.root = TrieNode()
    def add_word(self, word):
        node = self.root
        for ch in word:
            node = node.children.setdefault(ch, TrieNode())
        node.is_end = True
    def search(self, word):
        def dfs(node, i):
            if i == len(word):
                return node.is_end
            ch = word[i]
            if ch == '.':
                return any(dfs(child, i+1) for child in node.children.values())
            if ch not in node.children:
                return False
            return dfs(node.children[ch], i+1)
        return dfs(self.root, 0)
```
**TC:** add O(k); search O(k) average, O(26^k) worst case (all wildcards) **SC:** O(total chars)

### 73. Word Search II
Find all words from a list that exist in the grid (extends #49).
```python
# Brute: run Word Search (#49) independently for every word in the list
def find_words_brute(board, words):
    def exists(word):
        rows, cols = len(board), len(board[0])
        def dfs(r, c, i):
            if i == len(word):
                return True
            if r<0 or c<0 or r>=rows or c>=cols or board[r][c] != word[i]:
                return False
            temp, board[r][c] = board[r][c], '#'
            found = any(dfs(r+dr, c+dc, i+1) for dr, dc in [(-1,0),(1,0),(0,-1),(0,1)])
            board[r][c] = temp
            return found
        return any(dfs(r, c, 0) for r in range(rows) for c in range(cols))
    return [w for w in words if exists(w)]
```
**TC:** O(W · rows·cols·4^L) — full grid search per word **SC:** O(L) recursion stack
```python
# Optimal: build a Trie of all words, DFS the grid ONCE, pruning branches not in the Trie
class TrieNode:
    def __init__(self):
        self.children = {}
        self.word = None

def find_words(board, words):
    root = TrieNode()
    for w in words:
        node = root
        for ch in w:
            node = node.children.setdefault(ch, TrieNode())
        node.word = w

    rows, cols = len(board), len(board[0])
    result = set()

    def dfs(r, c, node):
        ch = board[r][c]
        if ch not in node.children:
            return
        nxt = node.children[ch]
        if nxt.word:
            result.add(nxt.word)
        board[r][c] = '#'
        for dr, dc in [(-1,0),(1,0),(0,-1),(0,1)]:
            nr, nc = r+dr, c+dc
            if 0<=nr<rows and 0<=nc<cols and board[nr][nc] != '#':
                dfs(nr, nc, nxt)
        board[r][c] = ch

    for r in range(rows):
        for c in range(cols):
            dfs(r, c, root)
    return list(result)
```
**TC:** O(rows·cols·4^L) total (shared Trie prunes dead branches early) **SC:** O(total chars in words)

### 74. Find Median from Data Stream
Support adding numbers and finding the median at any time.
```python
# Brute: keep a sorted list, insert with binary search, median = O(1) lookup once inserted
import bisect
class MedianFinderBrute:
    def __init__(self):
        self.data = []
    def add_num(self, num):
        bisect.insort(self.data, num)     # O(n) insertion (shifting elements)
    def find_median(self):
        n = len(self.data)
        mid = n // 2
        if n % 2:
            return self.data[mid]
        return (self.data[mid-1] + self.data[mid]) / 2
```
**TC:** add O(n), find_median O(1) **SC:** O(n)
```python
# Optimal: two heaps — max-heap for lower half, min-heap for upper half
import heapq
class MedianFinder:
    def __init__(self):
        self.small = []    # max-heap (negated)
        self.large = []    # min-heap
    def add_num(self, num):
        heapq.heappush(self.small, -num)
        heapq.heappush(self.large, -heapq.heappop(self.small))
        if len(self.large) > len(self.small):
            heapq.heappush(self.small, -heapq.heappop(self.large))
    def find_median(self):
        if len(self.small) > len(self.large):
            return -self.small[0]
        return (-self.small[0] + self.large[0]) / 2
```
**TC:** add O(log n), find_median O(1) **SC:** O(n)

### 75. Top K Frequent Elements
```python
# Brute: count frequencies, sort all of them
from collections import Counter
def top_k_frequent_brute(nums, k):
    freq = Counter(nums)
    return [val for val, _ in sorted(freq.items(), key=lambda x: -x[1])[:k]]
```
**TC:** O(n log n) **SC:** O(n)
```python
# Optimal: heap of size k (or bucket sort for O(n))
import heapq
from collections import Counter
def top_k_frequent(nums, k):
    freq = Counter(nums)
    return heapq.nlargest(k, freq.keys(), key=lambda x: freq[x])
```
**TC:** O(n log k) **SC:** O(n)

**General pattern to remember:** brute force almost always comes from nested loops (pairs/substrings/subsets)
or repeated linear scans (list membership, re-sorting, re-traversing). The optimal fix is nearly always
one of: a hashmap/set for O(1) lookups, sorting once + a linear pass, two pointers/sliding window,
a heap for "top k" or "streaming order", DP to cache overlapping subproblems, or a Trie/Union-Find for
structured lookups.
