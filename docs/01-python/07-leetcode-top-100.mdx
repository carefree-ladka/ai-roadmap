## Backtracking

### 1. Letter Combinations of a Phone Number
```python
# Brute: build combos by repeated string concatenation across all digits
def letter_combos_brute(digits):
    if not digits:
        return []
    m = {"2":"abc","3":"def","4":"ghi","5":"jkl","6":"mno","7":"pqrs","8":"tuv","9":"wxyz"}
    combos = [""]
    for d in digits:
        combos = [c + ch for c in combos for ch in m[d]]
    return combos
```
**TC:** O(4ⁿ · n) — same asymptotic as optimal, just built iteratively **SC:** O(4ⁿ) for output
```python
# Optimal: DFS backtracking, build one combo at a time
def letter_combos(digits):
    if not digits:
        return []
    m = {"2":"abc","3":"def","4":"ghi","5":"jkl","6":"mno","7":"pqrs","8":"tuv","9":"wxyz"}
    res, path = [], []
    def backtrack(i):
        if i == len(digits):
            res.append("".join(path))
            return
        for ch in m[digits[i]]:
            path.append(ch)
            backtrack(i + 1)
            path.pop()
    backtrack(0)
    return res
```
**TC:** O(4ⁿ · n) **SC:** O(n) recursion stack (excluding output)

### 2. Generate Parentheses
```python
# Brute: generate all 2^(2n) strings of length 2n, filter valid ones
def generate_parens_brute(n):
    def is_valid(s):
        bal = 0
        for ch in s:
            bal += 1 if ch == '(' else -1
            if bal < 0:
                return False
        return bal == 0
    res = []
    def gen(path):
        if len(path) == 2 * n:
            if is_valid(path):
                res.append(path)
            return
        gen(path + '(')
        gen(path + ')')
    gen("")
    return res
```
**TC:** O(2^(2n) · n) **SC:** O(2^(2n) · n)
```python
# Optimal: backtracking, only add ')' when valid — prunes invalid branches early
def generate_parens(n):
    res = []
    def backtrack(path, open_c, close_c):
        if len(path) == 2 * n:
            res.append("".join(path))
            return
        if open_c < n:
            path.append('('); backtrack(path, open_c+1, close_c); path.pop()
        if close_c < open_c:
            path.append(')'); backtrack(path, open_c, close_c+1); path.pop()
    backtrack([], 0, 0)
    return res
```
**TC:** O(4ⁿ / √n) — Catalan number bound **SC:** O(n) recursion stack (excluding output)

### 3. Combination Sum
Unique combinations (reuse allowed) summing to `target`.
```python
# Brute: generate all multisets up to a length bound, check sums
from itertools import combinations_with_replacement
def combination_sum_brute(candidates, target):
    res = set()
    max_len = target // min(candidates) if candidates else 0
    for length in range(1, max_len + 1):
        for combo in combinations_with_replacement(candidates, length):
            if sum(combo) == target:
                res.add(combo)
    return [list(c) for c in res]
```
**TC:** exponential, roughly O(C(target, len)) per length **SC:** O(result size)
```python
# Optimal: backtracking with pruning (sorted candidates, skip when sum exceeds target)
def combination_sum(candidates, target):
    candidates.sort()
    res, path = [], []
    def backtrack(start, remaining):
        if remaining == 0:
            res.append(path[:])
            return
        for i in range(start, len(candidates)):
            if candidates[i] > remaining:
                break
            path.append(candidates[i])
            backtrack(i, remaining - candidates[i])
            path.pop()
    backtrack(0, target)
    return res
```
**TC:** O(2^target) worst case, pruned in practice **SC:** O(target) recursion stack

**TC:** O(n! · n) **SC:** O(n! · n)
```python
# Optimal: backtracking, swap-in-place to avoid extra "used" tracking
def permute(nums):
    res = []
    def backtrack(start):
        if start == len(nums):
            res.append(nums[:])
            return
        for i in range(start, len(nums)):
            nums[start], nums[i] = nums[i], nums[start]
            backtrack(start + 1)
            nums[start], nums[i] = nums[i], nums[start]
    backtrack(0)
    return res
```
**TC:** O(n! · n) **SC:** O(n) recursion stack (excluding output)

### 5. N-Queens
```python
# Brute: try every placement of n queens across the board, validate at the end
from itertools import permutations
def solve_n_queens_brute(n):
    res = []
    for perm in permutations(range(n)):     # perm[col] = row for each column
        if len(set(perm[i]+i for i in range(n))) == n and len(set(perm[i]-i for i in range(n))) == n:
            board = ["." * perm[i] + "Q" + "." * (n-perm[i]-1) for i in range(n)]
            res.append(board)
    return res
```
**TC:** O(n! · n) — checks all permutations **SC:** O(n)
```python
# Optimal: backtracking with column/diagonal pruning sets
def solve_n_queens(n):
    res = []
    cols, diag1, diag2 = set(), set(), set()
    board = [["."] * n for _ in range(n)]
    def backtrack(r):
        if r == n:
            res.append(["".join(row) for row in board])
            return
        for c in range(n):
            if c in cols or (r-c) in diag1 or (r+c) in diag2:
                continue
            cols.add(c); diag1.add(r-c); diag2.add(r+c)
            board[r][c] = "Q"
            backtrack(r + 1)
            board[r][c] = "."
            cols.remove(c); diag1.remove(r-c); diag2.remove(r+c)
    backtrack(0)
    return res
```
**TC:** O(n!) with heavy pruning, much faster in practice than raw n! **SC:** O(n)

### 7. Word Search
*(Same as Blind 75 #49 — brute uses a visited set per call, optimal marks in place.)*
```python
def word_search_brute(board, word):
    rows, cols = len(board), len(board[0])
    def dfs(r, c, i, visited):
        if i == len(word):
            return True
        if r<0 or c<0 or r>=rows or c>=cols or (r,c) in visited or board[r][c] != word[i]:
            return False
        visited = visited | {(r, c)}
        return any(dfs(r+dr, c+dc, i+1, visited) for dr, dc in [(-1,0),(1,0),(0,-1),(0,1)])
    return any(dfs(r, c, 0, set()) for r in range(rows) for c in range(cols))
```
**TC:** O(rows·cols·4^L) plus O(L) set-copy overhead **SC:** O(L²)
```python
def word_search(board, word):
    rows, cols = len(board), len(board[0])
    def dfs(r, c, i):
        if i == len(word):
            return True
        if r<0 or c<0 or r>=rows or c>=cols or board[r][c] != word[i]:
            return False
        temp, board[r][c] = board[r][c], '#'
        found = any(dfs(r+dr, c+dc, i+1) for dr, dc in [(-1,0),(1,0),(0,-1),(0,1)])
        board[r][c] = temp
        return found
    return any(dfs(r, c, 0) for r in range(rows) for c in range(cols))
```
**TC:** O(rows·cols·4^L) **SC:** O(L) recursion stack

### 8. Palindrome Partitioning
All ways to partition `s` so every substring is a palindrome.
```python
# Brute: try all partitions (all subsets of "cut points"), filter valid ones
def partition_brute(s):
    n = len(s)
    res = []
    for mask in range(1 << (n - 1)):
        cuts = [0] + [i+1 for i in range(n-1) if mask & (1 << i)] + [n]
        parts = [s[cuts[i]:cuts[i+1]] for i in range(len(cuts)-1)]
        if all(p == p[::-1] for p in parts):
            res.append(parts)
    return res
```
**TC:** O(2ⁿ · n) **SC:** O(2ⁿ · n)
```python
# Optimal: backtracking, only recurse into palindromic prefixes
def partition(s):
    res, path = [], []
    def is_pal(sub):
        return sub == sub[::-1]
    def backtrack(start):
        if start == len(s):
            res.append(path[:])
            return
        for end in range(start + 1, len(s) + 1):
            if is_pal(s[start:end]):
                path.append(s[start:end])
                backtrack(end)
                path.pop()
    backtrack(0)
    return res
```
**TC:** O(2ⁿ · n) worst case, pruned in practice **SC:** O(n) recursion stack

### 9. Median of Two Sorted Arrays
```python
# Brute: merge both arrays, sort, pick the middle
def find_median_brute(nums1, nums2):
    merged = sorted(nums1 + nums2)
    n = len(merged)
    mid = n // 2
    return merged[mid] if n % 2 else (merged[mid-1] + merged[mid]) / 2
```
**TC:** O((n+m) log(n+m)) **SC:** O(n+m)
```python
# Optimal: binary search on the smaller array to find the correct partition
def find_median(nums1, nums2):
    if len(nums1) > len(nums2):
        nums1, nums2 = nums2, nums1
    n, m = len(nums1), len(nums2)
    lo, hi = 0, n
    while lo <= hi:
        i = (lo + hi) // 2
        j = (n + m + 1) // 2 - i
        left1 = nums1[i-1] if i > 0 else float("-inf")
        right1 = nums1[i] if i < n else float("inf")
        left2 = nums2[j-1] if j > 0 else float("-inf")
        right2 = nums2[j] if j < m else float("inf")
        if left1 <= right2 and left2 <= right1:
            if (n + m) % 2:
                return max(left1, left2)
            return (max(left1, left2) + min(right1, right2)) / 2
        elif left1 > right2:
            hi = i - 1
        else:
            lo = i + 1
```
**TC:** O(log(min(n, m))) **SC:** O(1)

```python
# Optimal: two binary searches (leftmost and rightmost bound)
```

```python
# Optimal: treat the matrix as a flattened sorted array, binary search
def search_matrix(matrix, target):
    if not matrix or not matrix[0]:
        return False
    rows, cols = len(matrix), len(matrix[0])
    lo, hi = 0, rows * cols - 1
    while lo <= hi:
        mid = (lo + hi) // 2
        val = matrix[mid // cols][mid % cols]
        if val == target:
            return True
        elif val < target:
            lo = mid + 1
        else:
            hi = mid - 1
    return False
```
**TC:** O(log(rows·cols)) **SC:** O(1)

### 14. Binary Tree Maximum Path Sum
*(Same as Blind 75 #63.)*
```python
def max_path_sum_brute(root):
    def gain(node):
        if not node:
            return 0
        return max(0, node.val + max(gain(node.left), gain(node.right)))
    def best_through(node):
        if not node:
            return float("-inf")
        return node.val + max(0, gain(node.left)) + max(0, gain(node.right))
    def all_nodes(node):
        if not node:
            return []
        return [node] + all_nodes(node.left) + all_nodes(node.right)
    return max(best_through(n) for n in all_nodes(root))
```
**TC:** O(n²) **SC:** O(h) + O(n)
```python
def max_path_sum(root):
    best = float("-inf")
    def gain(node):
        nonlocal best
        if not node:
            return 0
        l = max(gain(node.left), 0)
        r = max(gain(node.right), 0)
        best = max(best, node.val + l + r)
        return node.val + max(l, r)
    gain(root)
    return best
```
**TC:** O(n) **SC:** O(h)

### 16. Binary Tree Inorder Traversal
```python
# Brute: recursive, appends to a captured list (fine, but uses call stack heavily for skewed trees)
def inorder_brute(root):
    res = []
    def helper(node):
        if node:
            helper(node.left)
            res.append(node.val)
            helper(node.right)
    helper(root)
    return res
```
**TC:** O(n) **SC:** O(h) recursion stack (+ O(n) output)
```python
# Optimal: iterative with explicit stack — avoids recursion depth issues on skewed trees
def inorder(root):
    res, stack = [], []
    node = root
    while stack or node:
        while node:
            stack.append(node)
            node = node.left
        node = stack.pop()
        res.append(node.val)
        node = node.right
    return res
```
**TC:** O(n) **SC:** O(h) stack

```python
def is_valid_bst_brute(root):

        if node:
            inorder(node.left); vals.append(node.val); inorder(node.right)
```

```python
class Trie:
```

```python
        if nums[i] != i + 1:
            return i + 1
    return n + 1
```
**TC:** O(n) — each swap places at least one value correctly **SC:** O(1)

### 94. Merge Intervals
*(Same as Blind 75 #36.)*

```python
def merge_brute(intervals):
    intervals = [list(i) for i in intervals]
    changed = True
    while changed:
        changed = False
        for i in range(len(intervals)):
            for j in range(i+1, len(intervals)):
                a, b = intervals[i], intervals[j]
                if a[0] <= b[1] and b[0] <= a[1]:

    return intervals
```

```python
# Optimal: Dutch National Flag, one pass with three pointers
def sort_colors(nums):
    low, mid, high = 0, 0, len(nums) - 1
    while mid <= high:
        if nums[mid] == 0:
            nums[low], nums[mid] = nums[mid], nums[low]
            low += 1; mid += 1
        elif nums[mid] == 1:
            mid += 1
        else:
            nums[mid], nums[high] = nums[high], nums[mid]
            high -= 1
    return nums
```

### 98. Rotate Array

```python
def rotate_array_brute(nums, k):
```

### 100. Find the Duplicate Number
One duplicate in `1..n` range array of size `n+1`, without modifying the array or using extra space.
```python
# Brute: nested comparison
def find_duplicate_brute(nums):
    for i in range(len(nums)):
        for j in range(i+1, len(nums)):
            if nums[i] == nums[j]:
                return nums[i]
```
**TC:** O(n²) **SC:** O(1)
```python
# Optimal: Floyd's cycle detection treating the array as a linked list (value = next index)
def find_duplicate(nums):
    slow = fast = nums[0]
    while True:
        slow = nums[slow]
        fast = nums[nums[fast]]
        if slow == fast:
            break
    slow2 = nums[0]
    while slow != slow2:
        slow = nums[slow]
        slow2 = nums[slow2]
    return slow
```
**TC:** O(n) **SC:** O(1)

---

**Pattern recap:** most "brute force → optimal" jumps in this list fall into a handful of moves —
swap nested loops for a hashmap/set (O(n²)→O(n)), swap repeated sorting for one sort + linear pass
(O(n log n) done once instead of per-operation), swap linear scans for binary search on sorted data
(O(n)→O(log n)), swap recomputation for memoized/bottom-up DP (exponential→polynomial), or swap
extra data structures (visited sets, value arrays) for in-place pointer/index manipulation to cut SC to O(1).
